package org.bremersee.groupman.model;

import org.immutables.value.Value;
import tools.jackson.databind.annotation.JsonDeserialize;
import tools.jackson.databind.annotation.JsonSerialize;

/**
 * The group update model.
 *
 * @author Christian Bremer
 */
@Value.Style(
    visibility = Value.Style.ImplementationVisibility.PACKAGE,
    overshadowImplementation = true,
    depluralize = true,
    jdk9Collections = true,
    get = {"get*", "is*"},
    withUnaryOperator = "with*")
@Value.Immutable
@JsonSerialize(as = ImmutableGroupUpdate.class)
@JsonDeserialize(as = ImmutableGroupUpdate.class)
public interface GroupUpdate extends GroupBase {

  /**
   * Creates builder.
   *
   * @return the builder
   */
  static Builder builder() {
    return new Builder();
  }

  /**
   * The builder.
   */
  class Builder extends ImmutableGroupUpdate.Builder {

  }

}
