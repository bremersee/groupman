67    /**
68     * Instantiates a new group service.
69     *
70     * @param keycloakAdminClient the keycloak admin client
71     * @param groupValidation the group validation
72     * @param groupMapper the group mapper
73     * @param userMapper the user mapper
74     * @param realm the realm
75     * @param maxGroups the max groups (-1 == infinitive)
76     * @param mainGroupName the main group name
77     */
78    public GroupService(
79        KeycloakAdminClient keycloakAdminClient,
80        GroupValidation groupValidation,
81        GroupMapper groupMapper,
82        UserMapper userMapper,
83        @Value("${bremersee.keycloak.realm:master}") String realm,
84        @Value("${bremersee.groupman.max-groups:100}") int maxGroups,
85        @Value("${bremersee.groupman.main-group-name:groupman}") String mainGroupName) {
86  
87      Assert.notNull(keycloakAdminClient, "Keycloak admin client must not be null.");
88      Assert.notNull(groupValidation, "Group validation must not be null.");
89      Assert.notNull(groupMapper, "Group mapper must not be null.");
90      Assert.notNull(userMapper, "User mapper must not be null.");
91      Assert.hasText(realm, "Realm is required.");
92      this.keycloakAdminClient = keycloakAdminClient;
93      this.groupValidation = groupValidation;
94      this.groupMapper = groupMapper;
95      this.userMapper = userMapper;
96      this.realm = isEmpty(realm) ? "master" : realm;
97      this.maxGroups = maxGroups;
98      this.mainGroupName = isEmpty(mainGroupName) ? "groupman" : mainGroupName;
99    }
100 
101   /**
102    * Init.
103    */
104   @EventListener(ApplicationReadyEvent.class)
105   public void init() {
106     GroupRepresentation mainGroup = keycloakAdminClient
107         .getGroupByPath(realm, "/" + mainGroupName)
108         .singleOptional()
109         .flatMap(optMainGroup -> optMainGroup
110             .map(Mono::just)
111             .orElseGet(() -> keycloakAdminClient.saveGroup(realm, createNewMainGroup())))
112         .block();
113     if (isEmpty(mainGroup)) {
114       log.warn("Main group with name '{}' could not be created. Is keycloak admin api a MOCK?",
115           mainGroupName);
116     } else {
117       mainGroupId = mainGroup.getId();
118     }
119   }

121   /**
122    * Create group.
123    *
124    * @param userId the user id
125    * @param groupCreateRequest the group dto
126    * @return the mono
127    */
128   public Mono<Group> createGroup(String userId, GroupCreate groupCreateRequest) {
129     groupValidation.validateGroup(groupCreateRequest);
130     GroupRepresentation group = new GroupRepresentation();
131     groupMapper.mapInto(groupCreateRequest, group);
132     return getUserMainGroup(userId)
133         .filter(g -> maxGroups < 0
134             || maxGroups > requireNonNullElse(g.getSubGroupCount(), 0L))
135         .switchIfEmpty(Mono
136             .error(ServiceException.badRequest("Too many groups.", "too_many_groups")))
137         .mapNotNull(GroupRepresentation::getId)
138         .flatMap(id -> keycloakAdminClient.createSubGroup(realm, id, group))
139         .map(groupMapper::mapToDto);
140   }

142   /**
143    * Gets groups.
144    *
145    * @param userId the user id
146    * @param search the search
147    * @param first the first
148    * @param max the max
149    * @return the groups
150    */
151   public Flux<Group> getGroups(String userId, String search, Integer first, Integer max) {
152     var paramBuilder = GetGroupsParameters.builder()
153         .first(first != null && first >= 0 ? first : null)
154         .max(max != null && max > 0 ? max : null);
155     if (!isEmpty(search)) {
156       paramBuilder.search(search);
157     }
158     return getUserMainGroup(userId)
159         .mapNotNull(GroupRepresentation::getId)
160         .flatMapMany(groupId -> keycloakAdminClient
161             .getSubGroups(realm, groupId, paramBuilder.build()))
162         .map(groupMapper::mapToDto);
163   }

165   /**
166    * Gets group.
167    *
168    * @param userId the user id
169    * @param groupId the group id
170    * @return the group
171    */
172   public Mono<Group> getGroup(String userId, String groupId) {
173     return getOwnedGroup(userId, groupId)
174         .map(groupMapper::mapToDto);
175   }

177   /**
178    * Gets members.
179    *
180    * @param userId the user id
181    * @param groupId the group id
182    * @param first the first
183    * @param max the max
184    * @return the members
185    */
186   public Flux<User> getMembers(
187       String userId,
188       String groupId,
189       Integer first,
190       Integer max) {
191 
192     return Flux.from(getOwnedGroup(userId, groupId))
193         .flatMap(group -> keycloakAdminClient
194             .getGroupMembers(realm, groupId, null, first, max))
195         .map(userMapper::mapToDto);
196   }

198   /**
199    * Add member.
200    *
201    * @param userId the user id
202    * @param groupId the group id
203    * @param memberId the member id
204    * @return the mono
205    */
206   public Mono<Void> addMember(
207       String userId,
208       String groupId,
209       String memberId) {
210 
211     return getOwnedGroup(userId, groupId)
212         .flatMap(group -> keycloakAdminClient.addUserToGroup(realm, memberId, groupId));
213   }

218    * @param userId the user id
219    * @param groupId the group id
220    * @param memberId the member id
221    * @return the mono
222    */
223   public Mono<Void> removeMember(
224       String userId,
225       String groupId,
226       String memberId) {
227 
228     return getOwnedGroup(userId, groupId)
229         .flatMap(group -> keycloakAdminClient.removeUserFromGroup(realm, memberId, groupId));
230   }

232   /**
233    * Update group.
234    *
235    * @param userId the user id
236    * @param groupId the group id
237    * @param groupUpdateRequest the group update request
238    * @return the mono
239    */
240   public Mono<Group> updateGroup(
241       String userId,
242       String groupId,
243       GroupUpdate groupUpdateRequest) {
244 
245     groupValidation.validateGroup(groupUpdateRequest);
246     return getOwnedGroup(userId, groupId)
247         .flatMap(group -> {
248           groupMapper.mapInto(groupUpdateRequest, group);
249           return keycloakAdminClient.saveGroup(realm, group);
250         })
251         .map(groupMapper::mapToDto);
252   }

261   public Mono<Void> deleteGroup(String userId, String groupId) {

266   private GroupRepresentation createNewMainGroup() {
267     GroupRepresentation mainGroup = new GroupRepresentation();
268     mainGroup.setName(mainGroupName);
269     mainGroup.setDescription("Main group for groupman");
270     return mainGroup;
271   }
272 
273   /**
274    * Creates a new user main group representation without persisting.
275    *
276    * <p>The name of the group is the given user ID.
277    *
278    * @param userId the user ID
279    * @return the new user main group representation
280    */
281   private Mono<GroupRepresentation> newUserMainGroupRepresentation(String userId) {
282     return keycloakAdminClient.getUserById(realm, userId, null)
283         .map(user -> {
284           GroupRepresentation mainGroup = new GroupRepresentation();
285           mainGroup.setName(userId);
286           StringBuilder nameBuilder = new StringBuilder();
287           if (!isEmpty(user.getFirstName())) {
288             nameBuilder.append(user.getFirstName());
289           }
290           if (!isEmpty(user.getLastName())) {
291             if (!isEmpty(user.getFirstName())) {
292               nameBuilder.append(' ');
293             }
294             nameBuilder.append(user.getLastName());
295           }
296           if (!nameBuilder.isEmpty()) {
297             mainGroup.setDescription("Main group of " + nameBuilder);
298           }
299           return mainGroup;
300         })
301         .switchIfEmpty(Mono.error(ServiceException.forbidden("User", userId)));
302   }
303 
304   private Mono<GroupRepresentation> getUserMainGroup(String userId) {
305     if (isEmpty(mainGroupId)) {
306       return Mono.error(ServiceException
307           .internalServerError("Main group is not initialized.", "main_group_is_missing"));
308     }
309     return keycloakAdminClient.getGroupByPath(realm, "/" + mainGroupName + "/" + userId)
310         .singleOptional()
311         .flatMap(optUserMainGroup -> optUserMainGroup
312             .map(Mono::just)
313             .orElseGet(() -> newUserMainGroupRepresentation(userId)
314                 .flatMap(g -> keycloakAdminClient.createSubGroup(realm, mainGroupId, g))));
315   }
316 
317   private Mono<GroupRepresentation> getOwnedGroup(String userId, String groupId) {
318     return getUserMainGroup(userId)
319         .mapNotNull(GroupRepresentation::getId)
320         .flatMap(userMainGroupId -> keycloakAdminClient.getGroupById(realm, groupId)
321             .filter(group -> userMainGroupId.equals(group.getParentId())))
322         .switchIfEmpty(Mono.error(ServiceException
323             .notFoundWithErrorCode("Group", groupId, "group_not_found")));
324   }
